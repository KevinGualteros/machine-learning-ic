% 1. Calcular la matriz de confusión numéricamente

nombresClases = string(categories(YTrue));

% 2. Inicializar vectores
Precision   = zeros(numClasses, 1);
Recall      = zeros(numClasses, 1); % También llamado Sensibilidad
Specificity = zeros(numClasses, 1);
F1_Score    = zeros(numClasses, 1);

% 3. Calcular iterativamente para cada clase
for i = 1:numClasses
    TP = C(i, i);
    FP = sum(C(:, i)) - TP;
    FN = sum(C(i, :)) - TP;
    TN = sum(C(:)) - TP - FP - FN;
    
    % Precisión (Valor Predictivo Positivo)
    if (TP + FP) == 0
        Precision(i) = 0;
    else
        Precision(i) = TP / (TP + FP);
    end
    
    % Sensibilidad / Recall
    if (TP + FN) == 0
        Recall(i) = 0;
    else
        Recall(i) = TP / (TP + FN);
    end
    
    % Especificidad
    if (TN + FP) == 0
        Specificity(i) = 0;
    else
        Specificity(i) = TN / (TN + FP);
    end
    
    % F1-Score
    if (Precision(i) + Recall(i)) == 0
        F1_Score(i) = 0;
    else
        F1_Score(i) = 2 * (Precision(i) * Recall(i)) / (Precision(i) + Recall(i));
    end
end

% NUEVA FUNCIÓN: Aplica preprocesamiento Y transformaciones aleatorias
function dataOut = readAndAugmentData(data, globalClasses)
    % 1. Obtener la imagen base procesada
    dataOut_temp = readData(data, globalClasses);
    I = dataOut_temp{1};
    label = dataOut_temp{2};

    % 2. === AUMENTO DE DATOS ===
    % Espejo horizontal (50% probabilidad)
    if rand() > 0.5
        I = fliplr(I);
    end
    
    % Espejo vertical (50% probabilidad)
    if rand() > 0.5
        I = flipud(I);
    end
    
    % Rotación en múltiplos de 90 grados aleatoria (0, 90, 180, 270)
   % k = randi([0, 3]); 
   % I = rot90(I, k);

    % Devolver imagen aumentada
    dataOut = {I, label};
end

% FUNCIÓN ORIGINAL DE LECTURA (Usada para el Test sin alteraciones)
function dataOut = readData(data, globalClasses)
    imgPath = data{1}{1};
    roiPath = data{2}{1};
    lbl = data{3}{1};

    if ischar(lbl)
        lbl = string(lbl);
    end

    label = categorical(lbl, globalClasses);
    I = preprocessMammoROI(imgPath, roiPath);
    dataOut = {I, label};
end

% FUNCIÓN DE PREPROCESAMIENTO DE IMAGEN
function I = preprocessMammoROI(imgPath, roiPath)
    try
        % Leer imagen
        I = imread(imgPath);
        if size(I,3)==3
            I = rgb2gray(I);
        end
        I = imresize(I,[224 224]);

        % Leer ROI
        roi = imread(roiPath);
        if size(roi,3)==3
            roi = rgb2gray(roi);
        end
        roi = imresize(roi,[224 224]);
        roi = roi > 0;

        % Atenuar fondo
        I = im2double(I);
        I(roi == 0) = I(roi == 0) * 0.3;   

        % Normalización
        I = adapthisteq(I);
        I = im2single(I);
        I = repmat(I,1,1,3); % ResNet18 requiere 3 canales
    catch
        I = zeros(224,224,3,'single');
    end
end

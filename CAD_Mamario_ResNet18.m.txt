    function generarCaso(fig)
    
    app = fig.UserData;
    
    calc_test = app.calc_test;
    dicom_info = app.dicom_info;
    basePath = app.basePath;
    
    Ntest = height(calc_test);
    
    found = false;
    
    while ~found
    
    
    try
    
        idxRand = randi(Ntest);
    
        imgPathDICOM = ...
            calc_test.imageFilePath(idxRand);
    
        roiPathDICOM = ...
            calc_test.ROIPath(idxRand);
    
        imgID = extractBefore(imgPathDICOM,"/");
        roiID = extractBefore(roiPathDICOM,"/");
    
        rowIMG = ...
            dicom_info(dicom_info.PatientID==imgID,:);
    
        rowROI = ...
            dicom_info(dicom_info.PatientID==roiID,:);
    
        if isempty(rowIMG) || isempty(rowROI)
            continue
        end
    
        imgPath = fullfile( ...
            basePath,...
            "jpeg",...
            erase(string(rowIMG.image_path(1)),...
            "CBIS-DDSM/jpeg/"));
    
        roiPath = fullfile( ...
            basePath,...
            "jpeg",...
            erase(string(rowROI.image_path(1)),...
            "CBIS-DDSM/jpeg/"));
    
        if ~isfile(imgPath) || ~isfile(roiPath)
            continue
        end
    
        found = true;
    
    catch
        continue
    end
    
    
    end
    
    %% =====================================================
    % LEER IMAGEN ORIGINAL
    %% =====================================================
    
    I_original = imread(imgPath);
    
    if size(I_original,3)==3
    I_gray = rgb2gray(I_original);
    else
    I_gray = I_original;
    end
    
    I_gray = imresize(I_gray,[224 224]);
    
    %% =====================================================
    % LEER ROI
    %% =====================================================
    
    ROI = imread(roiPath);
    
    if size(ROI,3)==3
    ROI = rgb2gray(ROI);
    end
    
    ROI = imresize(ROI,[224 224]);
    
    %% =====================================================
    % PREPROCESAMIENTO
    %% =====================================================
    
    I_input = preprocessMammoROI( ...
    imgPath,...
    roiPath);
    

    app.currentImage = I_input;

app.currentGray = I_gray;
app.currentROI = ROI;

fig.UserData = app;
    
    %% =====================================================
    % CLASIFICACIÓN
    %% =====================================================
    
    tic
    
    [labelPred,scores] = ...
    classify(app.net,I_input);
    
    tiempoInferencia = toc;
    
    [probMax,~] = max(scores);
    
    %% =====================================================
    % GRAD-CAM
    %% =====================================================
    
 switch app.modeloActual

    case "resnet18"
        featureLayer = "res5b_relu";

    case "resnet50"
        featureLayer = "res5c_branch2c";

    case "densenet201"
        featureLayer = "conv5_block32_concat";

end
    
    
    gradcamMap = gradCAM( ...
    app.net,...
    I_input,...
    labelPred,...
    'FeatureLayer',...
    featureLayer);
    
    gradcamMap = ...
    imresize(gradcamMap,[224 224]);
    
    %% =====================================================
    % MOSTRAR IMAGEN ORIGINAL
    %% =====================================================
    
    imshow(I_gray,[],...
    'Parent',app.axOriginal);
    axis(app.axOriginal,'off')
axis(app.axROI,'off')
axis(app.axGradCAM,'off')
    
    title(app.axOriginal,...
    "Mamografía");
    
    %% =====================================================
    % MOSTRAR ROI
    %% =====================================================
    
    imshow(ROI,[],...
    'Parent',app.axROI);
    axis(app.axOriginal,'off')
axis(app.axROI,'off')
axis(app.axGradCAM,'off')
    
    title(app.axROI,...
    "ROI Real");
    
    %% =====================================================
    % MOSTRAR GRAD-CAM
    %% =====================================================
    
    imshow(I_gray,[],...
    'Parent',app.axGradCAM);
    axis(app.axOriginal,'off')
axis(app.axROI,'off')
axis(app.axGradCAM,'off')
    
    hold(app.axGradCAM,'on')
    
    imagesc(app.axGradCAM,...
    gradcamMap,...
    'AlphaData',0.5);
    
    colormap(app.axGradCAM,jet)
    
    hold(app.axGradCAM,'off')
    
    title(app.axGradCAM,...
    "Grad-CAM");
    
    %% =====================================================
    % PROBABILIDADES
    %% =====================================================
    
    cla(app.probAxes)
    
    barh(app.probAxes,...
    scores);
    
    app.probAxes.YTick = ...
    1:numel(app.classes);
    
    app.probAxes.YTickLabel = ...
    app.classes;
    
    xlim(app.probAxes,[0 1])
    
    title(app.probAxes,...
    "Probabilidades");
    
    %% =====================================================
    % INFORMACIÓN
    %% =====================================================
    
    app.lblClase.Text = ...
    "Clase: " + string(labelPred);
    
    app.lblConf.Text = ...
    sprintf( ...
    "Confianza: %.2f%%",...
    probMax*100);
    
    app.lblTiempo.Text = ...
    sprintf( ...
    "Tiempo: %.3f s",...
    tiempoInferencia);
    
    fig.UserData = app;
    
    end
    
    
    function I = preprocessMammoROI(imgPath, roiPath)
    
    try
    
    I = imread(imgPath);
    
    if size(I,3)==3
        I = rgb2gray(I);
    end
    
    I = imresize(I,[224 224]);
    
    roi = imread(roiPath);
    
    if size(roi,3)==3
        roi = rgb2gray(roi);
    end
    
    roi = imresize(roi,[224 224]);
    
    roi = roi > 0;
    
    I = im2double(I);
    
    I(roi==0) = ...
        I(roi==0)*0.3;
    
    I = adapthisteq(I);
    
    I = im2single(I);
    
    I = repmat(I,1,1,3);
    
    catch
    
    I = zeros(224,224,3,'single');
    
    
    end
    
    end
